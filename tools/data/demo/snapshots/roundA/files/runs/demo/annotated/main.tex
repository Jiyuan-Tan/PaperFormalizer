\begin{abstract}
We prove the two-variable inequality between the arithmetic and the geometric mean, and a cubic
inequality in three variables that follows from Schur's inequality.
\end{abstract}

\section{Preliminaries}

We use one classical result, which we cite without proof.

\begin{lemma}[Schur's inequality {\cite[Theorem~1]{hlp}}]\label{lem:schur}
For all real numbers $a, b, c \ge 0$,
\[
  a(a-b)(a-c) + b(b-a)(b-c) + c(c-a)(c-b) \ge 0 .
\]
\end{lemma}

\begin{lemma}\label{lem:sq}
For all real numbers $x$ and $y$, $x^2 + y^2 \ge 2xy$.
\end{lemma}

\begin{proof}
$x^2 + y^2 - 2xy = (x - y)^2 \ge 0$.
\end{proof}

\section{Main results}

\begin{theorem}\label{thm:amgm}
\vsOriginal{For all real numbers $a$ and $b$,
\[
  \frac{a + b}{2} \ge \sqrt{ab} .
\]} \vsNote{False for negative numbers: for $a=b=-1$ the left side is $-1$ while the right side is $\sqrt{1}=1$. The proof takes $\sqrt{a}$ and $\sqrt{b}$, which are real numbers only when $a,b\ge 0$. Counterexample: $a=b=-1$: $\frac{a+b}{2}=-1<1=\sqrt{ab}$. \vsLine \textbf{Suggested repair (not applied):} For all real numbers $a, b \ge 0$, \vsFit{\frac{a + b}{2} \ge \sqrt{ab} .}.}
\end{theorem}

\begin{proof}
Apply Lemma~\ref{lem:sq} to $x = \sqrt{a}$ and $y = \sqrt{b}$: then $x^2 = a$, $y^2 = b$ and
$xy = \sqrt{ab}$, so $a + b \ge 2\sqrt{ab}$.
\end{proof}

\begin{theorem}\label{thm:cubic}
For all real numbers $a, b, c \ge 0$,
\[
  (a + b + c)^3 + 9abc \ge 4(a + b + c)(ab + bc + ca) .
\]
\end{theorem}

\begin{proof}
Expanding both sides,
\[
  (a+b+c)^3 + 9abc - 4(a+b+c)(ab+bc+ca)
  = a^3 + b^3 + c^3 - \sum_{\mathrm{sym}} a^2 b + 3abc ,
\]
where $\sum_{\mathrm{sym}} a^2 b$ is the sum of the six terms $a^2 b$, $a^2 c$, $b^2 a$, $b^2 c$,
$c^2 a$, $c^2 b$. The right-hand side equals
$a(a-b)(a-c) + b(b-a)(b-c) + c(c-a)(c-b)$, which is nonnegative by Lemma~\ref{lem:schur}.
\end{proof}
